#### 执行命令
```shell
echo "开始构建ThingsBoard前端 (Angular + Yarn)..."

# 1. 进入前端项目目录
cd ui-ngx

# 2. 网络与Git环境配置（保留基础配置）
git config --global http.postBuffer 1048576000
# 可选：如果构建机有Git代理，可在此配置
# git config --global http.proxy http://your-proxy:port

# 3. 配置npm和安装构建工具
npm config set registry https://registry.npmmirror.com/
npm cache clean -f
npm install -g @angular/cli@latest yarn@latest

# 4. 验证环境
node -v
yarn -v

# 5. [核心修复] 修改yarn.lock，绕过镜像站缺失问题
echo "正在处理yarn.lock文件以修复Git依赖..."
if [ -f "yarn.lock" ]; then
    # 方案A：将github.com的https地址替换为git协议地址（通常更稳定）
    sed -i 's|https://github.com/|git://github.com/|g' yarn.lock
    # 方案B：如果方案A仍失败，可以尝试直接移除该特定依赖的resolution（如果非必需）
    # sed -i '/MichaelZinsmaier\\/CurvedLines/,/^$/d' yarn.lock
    echo "yarn.lock 文件已处理。"
else
    echo "警告：未找到 yarn.lock 文件。"
fi

# 6. 使用Yarn安装项目依赖（调整参数，去除--frozen-lockfile以便修改生效）
yarn install --network-timeout 600000 --retry 3

# 7. 执行构建
yarn run build

# 8. 验证产出
echo "构建完成，检查产出目录:"
ls -la target/generated-resources/public/
```

```shell
echo "begin..."

  echo "app_name can not be blank!"

  echo "app_new_build can not be blank!"
  exit 1

echo "app_new_build=${app_new_build}"
echo "app_new_build_path=${app_new_build_path}"
echo "env=${env}"

mkdir -p ${app_new_build_path}/${env}

tar zxvf ${app_new_build_path}/${env}/package.tgz -C ${app_new_build_path}/${env}

rm -rvf /usr/local/frontend/thingsboard-web-test/dist

cp -rvf ${app_new_build_path}/${env}/ui-ngx/target/generated-resources/public  /usr/local/frontend/thingsboard-web-test/dist

echo "end."
```

ui-ngx/bin/deploy.sh
```shell
#!/bin/bash  
#  
# 用于前端项目部署  
#  
  
PROG_NAME=$0  
usage() {  
  echo "Usage: $PROG_NAME [app_path] test|prod"  exit 2  
}  
  
if [ $# -lt 2 ]; then  
  usage  
  exit 1  
fi  
  
# app_new_build_path 应该为 /usr/local/youzhi-admin-new-build/binapp_new_build=$(dirname $0)  
app_new_build_path=$(  
  cd ${app_new_build}  
  pwd  
)  
  
app_path=$1  
env=$2  
  
echo "app_new_build=${app_new_build}"  
echo "app_new_build_path=${app_new_build_path}"  
echo "app_path=${app_path}"  
echo "env=${env}"  
  
if [ x"${app_path}" = x ]; then  
  echo "app_path can not be blank!"  
  usage  
  exit 1  
fi  
  
if [ x"${env}" = x ]; then  
  echo "env can not be blank!"  
  usage  
  exit 1  
fi  
  
# 创建出相关目录  
mkdir -p ${app_path}-${env}-tmp/  
mkdir -p ${app_path}-${env}-bak-new/  
mkdir -p ${app_path}-${env}-bak-old/  
  
rm -rf ${app_path}-${env}-bak-old  
cp -rf ${app_path}-${env}-bak-new ${app_path}-${env}-bak-old  
rm -rf ${app_path}-${env}-bak-new  
cp -rf ${app_path}-${env} ${app_path}-${env}-bak-new  
  
cp -rf ${app_new_build_path}/../dist ${app_path}-${env}-tmp/  
rm -rf ${app_path}-${env}  
mv ${app_path}-${env}-tmp ${app_path}-${env}  
  
test() {  
  echo "deploy test..."  
}  
prod() {  
  echo "deploy prod..."  
}  
  
case "${env}" in  
test)  
  test  
  ;;  
prod)  
  prod  
  ;;  
*)  
  usage  
  ;;  
esac
```
