    # tikhub_tiktok_scraper/tikhub_scraper/tikhub_scraper/pipelines.py
    import logging
    from sqlalchemy.orm import sessionmaker
    from tikhub_scraper.items import TikTokUserItem, TikTokUser
    import os
    from dotenv import load_dotenv
    
    class TikTokUserPipeline:
        def __init__(self):
            self.engine = None
            self.Session = None
            self.logger = logging.getLogger(__name__)
    
        @classmethod
        def from_crawler(cls, crawler):
            pipeline = cls()
            pipeline.crawler = crawler
            return pipeline
    
        def open_spider(self, spider):
            """爬虫开启时连接数据库"""
            load_dotenv()
    
            database_url = os.getenv('DATABASE_URL')
            if not database_url:
                self.logger.error("未找到DATABASE_URL环境变量")
                return
    
            try:
                from sqlalchemy import create_engine
                self.engine = create_engine(database_url)
                self.Session = sessionmaker(bind=self.engine)
                self.logger.info("数据库连接成功")
            except Exception as e:
                self.logger.error(f"数据库连接失败: {e}")
    
        def close_spider(self, spider):
            """爬虫关闭时断开数据库连接"""
            if self.engine:
                self.engine.dispose()
                self.logger.info("数据库连接已关闭")
    
        def process_item(self, item, spider):
            """处理爬取的数据项"""
            if not isinstance(item, TikTokUserItem):
                return item
    
            if not self.Session:
                self.logger.error("数据库会话未初始化")
                return item
    
            session = self.Session()
            try:
                # 检查用户是否已存在（基于user_id）
                existing_user = session.query(TikTokUser).filter_by(
                    user_id=item['user_id']
                ).first()
    
                if existing_user:
                    # 更新现有用户 - 更新所有字段
                    for field in item.fields:
                        if hasattr(existing_user, field) and field in item:
                            setattr(existing_user, field, item[field])
    
                    self.logger.info(f"更新用户数据: {item['unique_id']}")
                else:
                    # 创建新用户记录
                    user_data = {}
                    for field in item.fields:
                        if field in item:
                            user_data[field] = item[field]
    
                    user = TikTokUser(**user_data)
                    session.add(user)
                    self.logger.info(f"新增用户数据: {item['unique_id']}")
    
                session.commit()
    
            except Exception as e:
                session.rollback()
                self.logger.error(f"保存用户数据失败: {e}")
            finally:
                session.close()
    
            return item

    class JSONLogger:
        """JSON响应日志记录器"""
    
        def __init__(self, spider_name, log_dir="json_logs"):
            self.spider_name = spider_name
            self.log_dir = log_dir
            self.logger = logging.getLogger(__name__)

                os.makedirs(self.log_dir)

        def save_response(self, username, response_data, success=True):
            """保存API响应到JSON文件"""
            try:
                # 生成文件名
                timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
                status = "success" if success else "error"
                filename = f"{timestamp}_{username}_{status}.json"
                filepath = os.path.join(self.log_dir, filename)
    
                # 保存JSON文件
                with open(filepath, 'w', encoding='utf-8') as f:
                    json.dump(response_data, f, ensure_ascii=False, indent=2)
    
                self.logger.debug(f"已保存JSON响应: {filepath}")
                return filepath
    
            except Exception as e:
                self.logger.error(f"保存JSON响应失败: {e}")
                return None
    
        def save_error_response(self, username, error_info, request_url=None):
            """保存错误信息到JSON文件"""
            error_data = {
                "timestamp": datetime.now().isoformat(),
                "username": username,
                "error": str(error_info),
                "request_url": request_url,
                "success": False
            }
