## 一、原理与验证边界

```text
STL + 完整官方配置 → Bambu Studio CLI → G-code / 3MF → 本地语义与校验
→ 隐式FTPS:990 上传并读回比对 → MQTT TLS:8883 单次project_file
→ success → PREPARE → RUNNING → 现场确认正常打印
```

本次原执行链已实际上传、启动，并获用户确认正常打印；用户后续提供完成屏幕及取下的成品照片：完成、取件已确认，屏幕显示28分钟/4克，后续已获用户套入照片和“严丝合缝”反馈，记为初步合适；未量化公差或验证强度。下列切片与上传脚本由已执行块整理为独立完整脚本，启动脚本保持本次完整逻辑，仅把工作目录、IP及序列号改成环境变量。切片整理版已在临时目录重新运行、验证成功；约4.618 g/28分21秒，与原任务4.620 g/28分26秒略有差别，不宣称字节或估算完全一致。整理后的上传/启动脚本未再次向打印机执行。

本地临时输出没有覆盖正在打印的包，也没有重发启动。A1固件本次报告01.08.01.00；这些接口与不同机型、固件的兼容性不能据此保证。TLS使用设备自签名连接，与本次实现一致，仅在本人信任的局域网使用。

### 2.1 a1_prepare.py：只切片，不连接打印机

完整展开machine/process/filament的inherits与include；否则A1的启动模板、温度及打印板会丢失。只修复本次CLI的型号ID与异常可选首层时间元数据，不手改运动/温度G-code。

```python
"""Local slicing only: prepare one PETG Gauge_0; never connect to a printer."""
from pathlib import Path
import hashlib
import json
import os
import subprocess
import xml.etree.ElementTree as ET
import zipfile

app = Path(os.environ['BAMBU_APP_ROOT']).expanduser().resolve()
model = Path(os.environ['A1_MODEL']).expanduser().resolve()
job = Path(os.environ['A1_JOB_DIR']).expanduser().resolve()
params = job / '参数'
out = job / '切片'
for directory in [params, out, job / '启动记录']:
    directory.mkdir(parents=True, exist_ok=True)
filename = 'Gauge_0_A1_PETGBasic_20261103.gcode.3mf'
if (job / filename).exists():
    raise SystemExit('Use a new job directory; do not overwrite a verified package.')

def flatten_group(group, name):
    index = {}
    for path in (app / 'resources/profiles/BBL' / group).rglob('*.json'):
        data = json.loads(path.read_text())
        if data.get('name'):
            index.setdefault(data['name'], data)
    def merge(key, stack=()):
        if key in stack:
            raise ValueError('Profile inheritance cycle')
        data = index[key]
        result = merge(data['inherits'], stack + (key,)) if data.get('inherits') else {}
        for include in data.get('include', []):
            result.update(merge(include, stack + (key,)))
        result.update(data)
        result.pop('inherits', None)
        result.pop('include', None)
        return result
    return merge(name)

machine = flatten_group('machine', 'Bambu Lab A1 0.4 nozzle')
process = flatten_group('process', '0.20mm Standard @BBL A1')
filament = flatten_group('filament', 'Bambu PETG Basic @BBL A1')
process.update(curr_bed_type='Textured PEI Plate', layer_height='0.2',
    sparse_infill_density='15%', wall_loops='2', enable_support='0',
    initial_layer_speed=['20'], initial_layer_infill_speed=['30'],
    outer_wall_speed=['40'], inner_wall_speed=['60'], sparse_infill_speed=['60'],
    internal_solid_infill_speed=['60'], top_surface_speed=['40'],
    gap_infill_speed=['40'], bridge_speed=['30'], brim_type='no_brim')
filament.update(filament_colour=['#808080'], nozzle_temperature=['240'],
    nozzle_temperature_initial_layer=['240'], filament_max_volumetric_speed=['6'])
for name, data in [('machine.json', machine), ('process.json', process), ('filament.json', filament)]:
    (params / name).write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
env = os.environ.copy()
env.update(LD_LIBRARY_PATH=str(app / 'bin'), LC_ALL='C',
           SSL_CERT_FILE='/etc/ssl/certs/ca-certificates.crt')
command = [str(app / 'bin/bambu-studio'), '--datadir', str(job / 'cli-data'),
    '--load-settings', str(params / 'machine.json') + ';' + str(params / 'process.json'),
    '--load-filaments', str(params / 'filament.json'), '--arrange', '1', '--orient', '0',
    '--ensure-on-bed', '--slice', '0', '--outputdir', str(out), '--export-3mf', filename, str(model)]
with (job / '切片日志.txt').open('w') as log:
    subprocess.run(command, env=env, stdout=log, stderr=subprocess.STDOUT, check=True)
result = json.loads((out / 'result.json').read_text())
assert result['return_code'] == 0 and len(result['sliced_plates']) == 1
plate = result['sliced_plates'][0]
assert len(plate['objects']) == 1
box = plate['objects'][0]['bbox']
assert 0 <= box['x'] <= box['x'] + box['width'] <= 256
assert 0 <= box['y'] <= box['y'] + box['depth'] <= 256
model_id = json.loads((app / 'resources/profiles/BBL/machine/Bambu Lab A1.json').read_text())['model_id']
target = job / filename
with zipfile.ZipFile(out / filename) as source:
    assert source.testzip() is None
    gcode = source.read('Metadata/plate_1.gcode')
    digest = hashlib.md5(gcode).hexdigest()
    assert source.read('Metadata/plate_1.gcode.md5').decode().strip().lower() == digest
    metadata = ET.fromstring(source.read('Metadata/slice_info.config'))
    for item in metadata.findall('./plate/metadata'):
        if item.get('key') == 'printer_model_id':
            item.set('value', model_id)
        if item.get('key') == 'first_layer_time':
            item.set('value', '0.000000')
    with zipfile.ZipFile(target, 'w', compression=zipfile.ZIP_DEFLATED) as destination:
        for item in source.infolist():
            data = ET.tostring(metadata, encoding='utf-8', xml_declaration=True) if item.filename == 'Metadata/slice_info.config' else source.read(item.filename)
            destination.writestr(item, data)
with zipfile.ZipFile(target) as package:
    assert package.testzip() is None
    assert package.read('Metadata/plate_1.gcode') == gcode
    config = json.loads(package.read('Metadata/project_settings.config'))
    assert config['filament_type'] == ['PETG']
    assert config['nozzle_temperature'] == ['240'] and config['textured_plate_temp'] == ['70']
for line in gcode.decode().splitlines():
    if not line.lstrip().startswith(';'):
        assert '{' not in line and '}' not in line
manifest = dict(final_package=str(target), remote_file='/' + filename,
    final_package_sha256=hashlib.sha256(target.read_bytes()).hexdigest(), gcode_md5=digest,
    uploaded_to_printer=False, training_passed=False, physical_bed_clear=False,
    estimated_total_seconds=plate['total_predication'],
    estimated_filament_g=plate['filaments'][0]['total_used_g'])
(job / '准备核对结果.json').write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(manifest, ensure_ascii=False, indent=2))
```

### 2.2 a1_upload.py：只上传与读回校验，不启动

隐式FTPS端口990；本次设备需要TLS 1.2的数据通道会话复用。传输结束不额外unwrap，避免设备侧超时。返回226只表示上传结束，仍须读回SHA256相同。

```python
"""Upload/read-back verification only; this script never starts a print."""
import ftplib
import hashlib
import json
import os
from pathlib import Path
import ssl

class ImplicitTLS(ftplib.FTP_TLS):
    @property
    def sock(self):
        return self._sock
    @sock.setter
    def sock(self, value):
        if value is not None and not isinstance(value, ssl.SSLSocket):
            value = self.context.wrap_socket(value, server_hostname=self.host)
        self._sock = value
    def ntransfercmd(self, command, rest=None):
        connection, size = ftplib.FTP.ntransfercmd(self, command, rest)
        if self._prot_p:
            connection = self.context.wrap_socket(connection, server_hostname=self.host, session=self.sock.session)
        return connection, size
    def upload(self, command, file):
        self.voidcmd('TYPE I')
        with self.transfercmd(command) as connection:
            while block := file.read(65536):
                connection.sendall(block)
        return self.voidresp()
    def remote_hash(self, command):
        self.voidcmd('TYPE I')
        digest, size = hashlib.sha256(), 0
        with self.transfercmd(command) as connection:
            while block := connection.recv(65536):
                digest.update(block)
                size += len(block)
        self.voidresp()
        return size, digest.hexdigest()

job = Path(os.environ['A1_JOB_DIR']).expanduser().resolve()
manifest_path = job / '准备核对结果.json'
manifest = json.loads(manifest_path.read_text())
local = Path(manifest['final_package'])
context = ssl._create_unverified_context()
context.maximum_version = ssl.TLSVersion.TLSv1_2
ftp = ImplicitTLS(context=context, timeout=12)
ftp.connect(os.environ['BAMBU_IP'], 990)
ftp.login('bblp', os.environ['BAMBU_ACCESS_CODE'])
ftp.prot_p()
with local.open('rb') as file:
    print(ftp.upload('STOR /' + local.name, file))
size, digest = ftp.remote_hash('RETR /' + local.name)
assert size == local.stat().st_size and digest == hashlib.sha256(local.read_bytes()).hexdigest()
ftp.quit()
manifest.update(uploaded_to_printer=True, remote_verified_bytes=size, remote_verified_sha256=digest)
manifest_path.write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + '\n')
print('Uploaded package read-back SHA256 verified:', digest)
```

### 2.3 a1_start.py：默认只读状态，--start才启动

启动前要求当前清单中的现场/培训确认与已上传状态为真、设备空闲无错且外置料槽为PETG。同一启动记录存在时拒绝再次发送；网络超时后先查状态，不直接重试启动。

本次A1参数URL采用`file:///sdcard/文件名`，不能把其他机型的URL形式直接套用。`use_ams=false`；自动调平、流量与振动校准开启，延时摄影关闭。

```python
import datetime
import json
import os
from pathlib import Path
import secrets
import socket
import ssl
import struct
import sys
import time
from zoneinfo import ZoneInfo

ROOT = Path(os.environ['A1_JOB_DIR']).expanduser().resolve()
RECORD = ROOT / '启动记录' / '2026-11-03-PETG局域网启动记录.json'
SERIAL = os.environ['BAMBU_SERIAL']
FILENAME = 'Gauge_0_A1_PETGBasic_20261103.gcode.3mf'
START = '--start' in sys.argv
if START:
    manifest = json.loads((ROOT / '准备核对结果.json').read_text())
    if manifest.get('training_passed') is not True or manifest.get('physical_bed_clear') is not True:
        raise SystemExit('Required training and physical readiness confirmations are missing')
    if manifest.get('remote_file') != '/' + FILENAME or not manifest.get('uploaded_to_printer'):
        raise SystemExit('The verified PETG package is not the active uploaded task')

def stamp():
    return datetime.datetime.now(ZoneInfo('Asia/Shanghai')).isoformat(timespec='seconds')

def field(value):
    value = value.encode()
    return struct.pack('!H', len(value)) + value

def send(sock, kind, payload):
    header = bytearray([kind])
    size = len(payload)
    while True:
        byte = size % 128
        size //= 128
        header.append(byte | (128 if size else 0))
        if not size:
            break
    sock.sendall(header + payload)

def exact(sock, size):
    data = b''
    while len(data) < size:
        chunk = sock.recv(size - len(data))
        if not chunk:
            raise EOFError('Printer closed connection')
        data += chunk
    return data

def read(sock):
    kind = exact(sock, 1)[0]
    size, multiplier = 0, 1
    for _ in range(4):
        byte = exact(sock, 1)[0]
        size += (byte & 127) * multiplier
        if not byte & 128:
            break
        multiplier *= 128
    if size > 2000000:
        raise ValueError('Unexpected MQTT packet size')
    return kind, exact(sock, size)

def publish(sock, command):
    send(sock, 0x30, field('device/' + SERIAL + '/request') + json.dumps(command).encode())

def receive_print(sock):
    kind, data = read(sock)
    if kind >> 4 != 3:
        return {}
    topic_size = struct.unpack('!H', data[:2])[0]
    position = 2 + topic_size + (2 if (kind >> 1) & 3 else 0)
    return json.loads(data[position:]).get('print', {})

keys = ['gcode_state', 'subtask_name', 'gcode_file', 'mc_percent', 'mc_remaining_time',
        'stg_cur', 'mc_print_stage', 'layer_num', 'total_layer_num', 'nozzle_temper',
        'nozzle_target_temper', 'bed_temper', 'bed_target_temper', 'print_error', 'hms']
state = {}
record = {'created_at': stamp(), 'model': 'Bambu Lab A1', 'file': FILENAME, 'events': []}
if START and RECORD.exists():
    old = json.loads(RECORD.read_text())
    if old.get('start_command_attempted_at'):
        raise SystemExit('A start command was already attempted. Query status before any retry.')

context = ssl._create_unverified_context()
with context.wrap_socket(socket.create_connection((os.environ['BAMBU_IP'], 8883), timeout=5), server_hostname=SERIAL) as sock:
    sock.settimeout(3)
    send(sock, 0x10, field('MQTT') + bytes([4, 0xc2]) + struct.pack('!H', 20)
         + field('gauge-' + secrets.token_hex(4)) + field('bblp') + field(os.environ['BAMBU_ACCESS_CODE']))
    kind, data = read(sock)
    if kind != 0x20 or data[-1] != 0:
        raise RuntimeError('MQTT authentication failed')
    send(sock, 0x82, struct.pack('!H', 1) + field('device/' + SERIAL + '/report') + b'\0')
    read(sock)
    publish(sock, {'pushing': {'sequence_id': '51', 'command': 'pushall', 'version': 1, 'push_target': 1}})
    deadline = time.monotonic() + 14
    while time.monotonic() < deadline:
        try:
            state.update(receive_print(sock))
        except socket.timeout:
            continue
        if 'gcode_state' in state and 'print_error' in state:
            break
    print('Initial status:', json.dumps({k:state[k] for k in keys if k in state}, ensure_ascii=False), flush=True)
    if not START:
        raise SystemExit(0)
    if state.get('gcode_state') != 'IDLE' or state.get('print_error') != 0:
        raise SystemExit('Printer is not idle without errors; no start command sent')
    if state.get('vt_tray', {}).get('tray_type') != 'PETG':
        raise SystemExit('Printer external-spool metadata is not PETG; no start sent')
    sequence = str(int(time.time()))
    command = {'print': {'sequence_id': sequence, 'command': 'project_file',
        'param': 'Metadata/plate_1.gcode', 'url': 'file:///sdcard/' + FILENAME,
        'bed_type': 'auto', 'timelapse': False, 'bed_leveling': True,
        'flow_cali': True, 'vibration_cali': True, 'layer_inspect': False,
        'use_ams': False, 'ams_mapping': [0], 'subtask_name': FILENAME,
        'profile_id': '0', 'project_id': '0', 'subtask_id': '0', 'task_id': '0'}}
    record.update(start_command_attempted_at=stamp(), command=command,
                  physical_bed_confirmation='User confirmed installed and empty, sufficient PETG, nozzle clear and waste bin not full before startup')
    RECORD.write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n')
    publish(sock, command)
    print('One start command sent; sequence:', sequence, flush=True)
    deadline = time.monotonic() + 100
    next_push = time.monotonic() + 12
    last_display = None
    last_print = 0
    while time.monotonic() < deadline:
        if time.monotonic() >= next_push:
            publish(sock, {'pushing': {'sequence_id': '52', 'command': 'pushall', 'version': 1, 'push_target': 1}})
            next_push = time.monotonic() + 15
        try:
            change = receive_print(sock)
        except socket.timeout:
            send(sock, 0xc0, b'')
            continue
        state.update(change)
        if change.get('command') == 'project_file':
            ack = {k:v for k,v in change.items() if k in ['command','sequence_id','result','reason','err_code']}
            record['events'].append({'at':stamp(),'ack':ack})
            print('Print acknowledgement:', json.dumps(ack, ensure_ascii=False), flush=True)
        snapshot = {k:state[k] for k in keys if k in state}
        if snapshot != last_display and time.monotonic() - last_print >= 5:
            record['events'].append({'at':stamp(),'status':snapshot})
            RECORD.write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n')
            print(stamp(), json.dumps(snapshot, ensure_ascii=False), flush=True)
            last_display = snapshot
            last_print = time.monotonic()
    record['latest_status'] = {k:state[k] for k in keys if k in state}
    record['observation_ended_at'] = stamp()
    RECORD.write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n')
```

## 五、踩坑记录

| 现象 | 原因 → 解决 |
| --- | --- |
| 切成200°C或其他板温，且无A1正确开场 | 只导入leaf JSON → 完整展开inherits/include及machine模板后重新切片。 |
| 导出路径被拼成两段绝对路径 | 已用--outputdir又给绝对--export-3mf → export只传文件名。 |
| GLEW/OpenGL失败但切片返回0 | 无头缩略图生成失败 → 不将“无缩略图”当作没切片；另核result、模型、G-code和CRC。不能据此宣称已验证图形渲染。 |
| 首层时间是异常大数、型号ID为空 | 本次CLI导出元数据异常 → 从官方A1模型取N2S，将可选首层时间与CLI plate JSON的0一致；不改G-code。 |
| FTPS控制通道通、数据通道卡住 | 隐式TLS及会话复用差异 → 本次用TLS1.2复用控制会话；保留读回校验。 |
| 摄像头有响应却看不清 | 返回模糊画面 → 不能作为清空平台或首层通过证据，需现场确认。 |
| 程序显示RUNNING就当模型已出料 | RUNNING也包括加热/校准 → 记录阶段，另以现场观察确认正常打印。 |
| 新切片包与旧包hash不同 | 时间戳/元数据及配置整理会影响字节 → 核对每个新包的语义和自身MD5/SHA256，不套用旧包hash。 |
